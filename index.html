    // ★ 1단계에서 얻은 구글 Apps Script URL을 아래 큼지막한 따옴표 안에 넣으세요.

    const APPS_SCRIPT_URL = "여기에_구글_APPS_SCRIPT_URL을_붙여넣으세요";

    let experimentData = {

      choice1: "",

      choice2: "",

      hesitationTime: 0,

      justification: 4,

      responsibility: 4

    };

    let startTime = 0;

    function submitChoice1(choice) {

      experimentData.choice1 = choice;

      document.getElementById('page1').classList.remove('active');

      document.getElementById('page2').classList.add('active');

     

      // 2단계 페이지 진입 시점 기록 (고민 시간 측정 시작)

      startTime = performance.now();

    }

    function submitChoice2(choice) {

      // 선택 완료 시점 - 진입 시점 = 고민 시간 (밀리초 단위)

      let endTime = performance.now();

      experimentData.hesitationTime = Math.round(endTime - startTime);

      experimentData.choice2 = choice;

      document.getElementById('page2').classList.remove('active');

      document.getElementById('page3').classList.add('active');

    }

    function sendData() {

      experimentData.justification = document.getElementById('justification').value;

      experimentData.responsibility = document.getElementById('responsibility').value;

      fetch(APPS_SCRIPT_URL, {

        method: "POST",

        mode: "no-cors",

        headers: { "Content-Type": "application/json" },

        body: JSON.stringify(experimentData)

      }).then(() => {

        document.getElementById('page3').classList.remove('active');

        document.getElementById('page4').classList.add('active');

      }).catch(err => alert("전송 에러: " + err));

    }
